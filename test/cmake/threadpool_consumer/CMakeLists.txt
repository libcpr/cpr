cmake_minimum_required(VERSION 3.24)
project(cpr_threadpool_consumer LANGUAGES CXX)
find_package(cpr CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_compile_features(consumer PRIVATE cxx_std_17)
target_link_libraries(consumer PRIVATE cpr::cpr)
if(CPR_TEST_CUSTOM_CONTEXT)
    find_package(Boost 1.70 REQUIRED)
    target_compile_definitions(consumer PRIVATE CPR_TEST_CUSTOM_CONTEXT)
    target_link_libraries(consumer PRIVATE Boost::headers)
endif()
enable_testing()
add_test(NAME consumer COMMAND consumer)
